# Pipeline da Fábrica Local de Reels (Pillow + FFmpeg + Edge-TTS)
def render_reel(script_json, resolution=(1080, 1920)):
    acts = script_json["acts"]  # Roteiro em 7 atos estruturados
    # 1. Síntese neural pt-BR sincronizada
    audio_tracks = synthesize_voice(acts, voice="pt-BR-AntonioNeural")
    # 2. Renderização rápida do canvas 720p
    raw_frames = render_canvas_frames(acts, size=(720, 1280))
    # 3. Interpolação Lanczos para nitidez mobile máxima
    upscaled = lanczos_upscale(raw_frames, target=resolution)
    # 4. Folha de contato para validação humana em 1 clique
    generate_contact_sheet(upscaled, output="contact_sheet.jpg")
    return compile_mp4(upscaled, audio_tracks)
